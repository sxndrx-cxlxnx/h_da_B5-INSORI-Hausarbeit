\chapter*{Abschrift der Mitschrift zur Diskussionsrunde beim Kolloquium}
\addcontentsline{toc}{section}{Abschrift der Mitschrift zur Diskussionsrunde beim Kolloquium}
\label{MitschriftKolloquium}

\noindent \textbf{Frage:} \textit{Allgemeine Meinung zur Chatkontrolle - Wie viel Privatsphäre bist Du bereit aufzugeben, um das Ziel zu erreichen?}\\
\underline{Pro:}
\begin{itemize}
    \item Zweck heiligt die Mittel $\rightarrow$ Privatsphäre wäre es wert Kinder / kleine Geschwister zu schützen
    \item \glqq ich habe nichts zu verbergen\grqq{}
    \item Überwachung gibt ein Sicherheitsgefühl
\end{itemize}
\underline{Kontra:}
\begin{itemize}
    \item es ist ein drastisches Eingreifen in die Privatsphäre
    \begin{itemize}
        \item Betroffene versuchen die Kontrolle zu umgehen (externe Verschlüsselung)
        \item damit wurde das Ziel verfehlt, weil eine Alternative Kommunikationsweise gefunden wurde
    \end{itemize}
    \item Hemmschwelle
    \begin{itemize}
        \item Wenn Chatkontrolle in diesem Bereich eingeführt wird, kann das woanders eher auch zugelassen werden
        \item Dammbruchargument $\rightarrow$ = Problem
    \end{itemize}
    \item Achsen reformulieren: Universalismus vs. Partikularismus (Note: entsprechend zum Diskussionskapitel $\rightarrow$ rausnehmen?)
\end{itemize}

% *******************************************************************************************

\noindent \textbf{Frage:} \textit{Ist es dir wichtiger, dass alle CSAM gefunden werden, oder ist die Gefahr zu groß, falsch verdächtigt zu werden?}\\
\underline{Kontra:}
\begin{itemize}
    \item wenn kritische Strukturen bei Einzelpersonen gefunden werden, ist Chatkontrolle ok $\rightarrow$ trotz kritischem Vorwand [die Chatkontrolle auf alle angewandt ist demnach nicht in Ordnung]
\end{itemize}

% *******************************************************************************************

\noindent \textbf{Frage:} \textit{Habt ihr noch weitere Ideen, wie das Ziel erreicht werden kann?}\\
\underline{Kontra:}
\begin{itemize}
    \item andere Wege für Kontrolle, z.B. durch die Eltern
    \begin{itemize}
        \item Eltern kontrollieren Social Media
        \item ABER: ist trotzdem ein Eingriff in die Privatsphäre der Kinder, hier aber von den Eltern
    \end{itemize}
\end{itemize}
